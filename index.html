async function confirmCompleteTrip(e) { e.preventDefault(); const id = document.getElementById('modalTripId').value; // جلب البيانات الأساسية للرحلة مباشرة لضمان دقة الأرقام المالية const { data: tripData } = await client.from('trips').select('total_amount, paid_amount, status, truck_plate, driver_name').eq('id', id).single(); if (!tripData) return; const total = parseFloat(tripData.total_amount || 0); const currentPaid = parseFloat(tripData.paid_amount || 0); let additionalPayment = parseFloat(document.getElementById('modalPaymentInput').value) || 0; const finalCommission = parseFloat(document.getElementById('modalDriverCommission').value) || 0; let newPaidAmount = currentPaid + additionalPayment; if (newPaidAmount > total) newPaidAmount = total; const newRemaining = total - newPaidAmount; const payment_status = newRemaining <= 0 ? 'مسدد بالكامل' : 'سداد جزئي'; const updateData = { paid_amount: newPaidAmount, remaining_amount: newRemaining > 0 ? newRemaining : 0, payment_status: payment_status, driver_commission: finalCommission, status: 'مكتملة' }; const { error } = await client.from('trips').update(updateData).eq('id', id); if (!error) { if (tripData.truck_plate) { await client.from('trucks').update({ status: 'جاهزة للعمل' }).eq('plate', tripData.truck_plate); } if (tripData.driver_name) { await client.from('drivers').update({ status: 'متوفر' }).eq('name', tripData.driver_name); } closeCompleteModal(); alert('تم تسجيل السداد والعمولة وإنهاء الرحلة بنجاح!'); loadAllData(); } else { alert('حدث خطأ أثناء حفظ التسوية.'); } }
